    \uline{\textbf{Lecture 1 - Software Dev Processes}}

    \textbf{Software engineering} is the application of a systematic, disciplined, quantifiable approach
    to the development, operation, and maintenance of software

    \begin{itemize}
        \item Push factors
        \begin{itemize}
            \item Hardware advances + cost reduction
            \item Data availability + device proliferation
            \item Increases in processing power
        \end{itemize}
        \item Pull factors
        \begin{itemize}
            \item Digital natives (consuming technology)
        \end{itemize}
        \item Responses: R\&D in 
        \begin{itemize}
            \item Parallel processing
            \item Cloud computing
            \item Data analytics
            \item Security
            \item IoT
            \item Edge computing
            \item Devops, CI, CD
            \item Microservices
        \end{itemize}

    \textbf{Software delivery and deployment}
    
    Activities that make the software available for use after the development

    \textbf{Deployment:} process between acquisition of software and execution of software

    Deployment decisions can affect the quality attributes

    \textbf{Deployment considerations}

    \begin{itemize}
        \item Integration of the internet and related advances
        \begin{itemize}
            \item Virtual worldwide marketplace
            \item Considerations: shift towards cloud-nativity
            \item \textit{Quality attribute: Portability}
        \end{itemize}
        \item Large-scale content delivery
        \begin{itemize}
            \item Transfer data/information between various end-points
            \item Considerations: location, number of end-points, quality of service
            \item \textit{Quality attribute: Availability, performance}
        \end{itemize}
        \item Heterogenous platforms
        \begin{itemize}
            \item Variety of OS/Browser/Platform combinations
            \item Considerations: Coexistence of platforms and interoperability between platforms
            \item \textit{Quality attribute: Interoperability}
        \end{itemize}
        \item Dependency and change management
        \begin{itemize}
            \item Multiple components have interdependencies 
            \begin{itemize}
                \item ``uses'' relationship
            \end{itemize}
            \item Considerations: configurations, version changes (within and outside components), compatibility
            \item \textit{Quality attribute: Maintainability}
        \end{itemize}
        \item Coordination \& communication among components
        \begin{itemize}
            \item Computations among components need coordination
            \item Considerations: API design, nature (synchronous/async) of communication
            \item \textit{Quality attribute: Performance}
        \end{itemize}
        \item Security
        \begin{itemize}
            \item Requirements of privacy, authentication, authorization, and integrity
            \item Considerations: mechanisms for encryption, authentication \& authorization management, attribution
            \item \textit{Quality attribute: Security, usability}
        \end{itemize}
    \end{itemize}

    \textbf{Deployment mechanisms}
    \begin{Figure}
       \centering
       \includegraphics[width=\linewidth]{deployment_mechanisms.png}
    \end{Figure}

    \uline{Bare metal}
    
    \begin{itemize}
        \item From code to executable
        \begin{itemize}
            \item Catering to target platforms
            \item Customized build \& linking
            \item Factors affecting: availability of libraries \& dependencies 
        \end{itemize}
        \item Potentially wasted hardware resources
        \item Cost factor
        \item Scalability issues
        \item Developer productivity reduced
        \item Benefits:
        \begin{itemize}
            \item Complete control
            \item Physical isolation
        \end{itemize}
    \end{itemize}

    \uline{Virtual machines}
    
    \begin{itemize}
        \item VMs improve resource utilization $\rightarrow$ reduced cost
        \item Flexible, scalable
        \item Problems:
        \begin{itemize}
            \item Still a full OS running inside VMs
            \item Susceptible to side-channel attacks
            \item ``Noisy neighbor'' problem (if 1 VM uses too much memory, it will affect other VMs)
        \end{itemize}
    \end{itemize}

    \uline{Containers}
    
    \begin{itemize}
        \item Benefits:
        \begin{itemize}
            \item Lighter than VMs
            \begin{itemize}
                \item Include only OS processes and dependencies necessary
                \item Better utilization of hardware capacity
                \item Allows rapid deployment
            \end{itemize}
            \item Write once run anywhere
            \begin{itemize}
                \item Provision containers to run the applications
                \item Ideal for CI/CD, Agile/DevOps practices
            \end{itemize}
            \item Granular \& controllable
            \begin{itemize}
                \item Deployments can be of the whole system or of elements within a system
                \item Monitor deployment, rollback / patch / redeploy easily
            \end{itemize}
            \item Easy integration of the internet and related advances
            \begin{itemize}
                \item Build cloud-native applications
            \end{itemize}
            \item Include runtime with code
            \begin{itemize}
                \item Caters to heterogeneous platforms
                \item Can achieve interoperability \& portability
            \end{itemize}
            \item Supports dependency and change management
            \begin{itemize}
                \item Improved maintainability of the software
            \end{itemize}
            \item Environment management
            \begin{itemize}
                \item Applies to both development \& deployment
                \item E.g., Dockerfile provides the environment description
            \end{itemize}
            \item Reproducible
            \begin{itemize}
                \item Guaranteed to be identical on any system that can run containers
            \end{itemize}
            \item Isolation \& security
            \begin{itemize}
                \item Avoid conflicting dependencies
                \item Can provide some sand-boxing for code execution
            \end{itemize}
            \item Quick to launch (order of milliseconds)
            \item Supports DevOps best practices
            \item Can be used with orchestrator like Kubernetes
            \begin{itemize}
                \item Orchestration automates provisioning, deployment, networking, scaling, availability, and lifecycle management of containers
                \item Decides how and where containers run
            \end{itemize}
    \end{itemize}

    \item Problems:
        \begin{itemize}
            \item Not suitable for all applications
            \item Not suitable for performance-critical applications
        \end{itemize}
    \end{itemize}

    \textbf{Container vs Orchestrator}

    \textbf{Containers} provide the platform for building and distributing services
    \begin{itemize}
        \item Not suited for running complex applications on its own
        \item Often need multiple containers, each (set) doing specific tasks / providing specific services
    \end{itemize}

    An \textbf{orchestrator} is a separate software that:
    \begin{itemize}
        \item Integrates and coordinates many parts (modules, hosted on containers),
        \item scales up / down the deployment based on demand,
        \item provides fault tolerance to the application, and
        \item provides communication among the containers.
    \end{itemize}

    Orchestrator works with the containers to realize the production. (e.g. container: docker, orchestrator: kubernetes)

    \uline{Serverless}
    \begin{itemize}
        \item A cloud-native development model
        \item There are still serveres, but developers / enterprises don't have to manage the servers
        \item Developers package the code in containers
        \begin{itemize}
            \item Application deployed via these containers
            \item Once deployed, apps respond to demand - automatically scale up / down as needed
            \item Economics: Metered on demand $\Rightarrow$ no cost when idle
        \end{itemize}
        \item Example: AWS Lambda
        \item Cloud provider manages the physical servers
        \item Dynamically allocate resources on behalf of developers (or users)
        \item Code deployed to production
        \item Works based on Function-As-A-Services (FAAS)
        \begin{itemize}
            \item Event-driven execution model (run when needed)
            \item App logic deployed in containers
            \item Such containers are managed by the platform / service provider
            \item Stateless applications
            \item Ephemeral - short execution times
        \end{itemize}
        \item Example applications:
        \begin{itemize}
            \item Batch processing of image and video
            \item Scheduled data compression
        \end{itemize}
    \end{itemize}

    \textbf{Software Delivery}

    \textbf{Deployment pipeline} is an automated implementation of application build, deploy, test, and release process
    
    \textbf{CI/CD pipeline} is an automated workflow that enables teams to build, test, and deploy
    code efficiently and reliably

    \uline{Continuous integration}

    Build $\Rightarrow$ Unit Tests $\Rightarrow$ Deploy to stage $\Rightarrow$ Acceptance tests

    \uline{Continuous delivery}

    Build $\Rightarrow$ Unit Tests $\Rightarrow$ Deploy to stage $\Rightarrow$ Acceptance tests $\Rightarrow$ (manually) Deploy to production

    \uline{Continuous deployment}

    Build $\Rightarrow$ Unit Tests $\Rightarrow$ Deploy to stage $\Rightarrow$ Acceptance tests $\Rightarrow$ (automatically) Deploy to production

    \textbf{Continuous delivery and deployment}

    The ability to get changes of all types (new features, config changes, bug fixes, experiments) into production,
    or into the hands of users, safely and quickly in a sustainable way.

    Benefits:
    \begin{itemize}
        \item Low-risk releases (human error in the process of deploying is reduced, small portions
        delivered continuously and can get feedback from consumers)
        \item Faster time to market and early feedback
        \item Higher quality (continuous feedback $\rightarrow$ detect bugs earlier and fix earlier)
        \item Lower cost (saving man-hours due to automation, bugs late in the cycle are expensive)
    \end{itemize}

    \textbf{Continuous delivery}: Software \uline{can be} released into prod at any given time

    \textbf{Continuous deployment}: Software is \uline{released automatically} into prod

    \textbf{Strategies for Continuous Deployment}

    \uline{In-place deployment}         

    \begin{itemize}
        \item Each module/compute is stopped, latest version installed, started, \& validated
        \item Minimal disturbance to the infrastructure
        \item Reduced infrastructure cost
        \item Affects availability (bc of downtime when u shut down the old version \& bring up the new one)
        \item E.g., AWS Beanstalk allows one-at-a-time, half-at-a-time, all-at-once mechanisms for deployment
    \end{itemize}

    \uline{Rolling deployment}

    \begin{itemize}
        \item Replace previous versions of the application by completely replacing the
        infrastructure on which the application is running
        \item E.g., containers running previous versions of the application will be replaced one-by-one
        with containers running new versions of the application
        \item No isolation between old/new versions
        \item Complicates rollback
        \item Some using old version, some using new version
    \end{itemize}

    \begin{Figure}
       \centering
       \includegraphics[width=0.8\linewidth]{rolling_deployment.png}
    \end{Figure}

    \uline{Blue/Green deployment}
    \begin{itemize}
        \item Shift traffic between 2 identical environments
        \item Run different versions of the application in each
        \item Near-zero downtime
        \item Allows rollback
        \item Simplified CI/CD workflow by limiting the complexity
    \end{itemize}
    Example: Amazon e-commerce

    \begin{Figure}
       \centering
       \includegraphics[width=\linewidth]{blue_green.png}
    \end{Figure}

    \uline{Canary deployment}

    \begin{itemize}
        \item Reduce the risk of deployment
        \item Incrementally deploy new versions, starting with a small subset of users
        \item AKA incremental rollout
        \item Can be used for A/B testing
    \end{itemize}
    E.g. Google Chrome Canary

    \textbf{DevOps}

    DevOps is a set of software development practices that combine software development and operations.
    The set of practices intends to reduce the time between committing a change to a system, and the change
    being placed into normal production while ensuring high quality.

    \textbf{DevOps pipeline} is a set of automated processes and tools that allows developers
    and operations professionals to collaborate on building and deploying code to a production environment.

    \textbf{Benefits of DevOps}

    \begin{itemize}
        \item \textbf{Speed of delivery}: The model enables the dev and ops teams to deliver faster
        for customers. Practices of CI/CD that automate the software release process, from build to deploy
        can quicker release new features and fix bugs.
        \item \textbf{Reliability}: CI/CD ensure testing that each change is functional and safe.
        Monitoring and logging help stay informed of performance in real-time. These contribute
        to more reliable delivery and positive experience of end-users.
        \item \textbf{Scale}: Operating and managing infrastructure and development processes with 
        automation and consistency help improve scalability. Infrastructure as code helps manage dev, testing,
        and prod envs in a repeatable and more efficient manner.
        \item \textbf{Improved collaboration}: Dev and ops teams collaborate closely, share responsibilities, and workflows.
        This reduces inefficiencies and saves time (e.g. reduced handover periods between dev and ops)
    \end{itemize}

    \uline{\textbf{Lecture 2 - Modern Software Engineering}}

    \textbf{Automation}
    \begin{itemize}
        \item Repeated tasks
        \item Know what needs to be document
        \item AI: automate, do it efficiently and reliably, every time
        \item E.g. workflow automation (CI/CD, infrastructure deployment, ...)
    \end{itemize}

    \textbf{Pattern detection}
    \begin{itemize}
        \item Tasks adhere to some rules
        \item Exploit the structure
        \item AI: sift through gigantic codebase to detect vulnerabilities
        \item E.g. identifying query patterns
    \end{itemize}

    \uline{Example: vulnerability detection}

    Observation: vulnerabilities are caused by a single repeating pattern $\rightarrow$ solution is
    to exploit ML (+ expert knowledge) to identify such patterns

    Mechanism components:
    \begin{itemize}
        \item Encoded expert knowledge
        \item Large volumes of weak evidence
        \item Pattern detection
        \item Deductive and inductive reasoning
        \item Create higher-level features
    \end{itemize}

    \textbf{Collaboration}
    \begin{itemize}
        \item Code sequence predicition and generation
        \item Brainstorming code solutions
        \item Rapid prototyping
        \item E.g. Copilot, ChatGPT, ...
        \item Turn-by-turn development
        \item AI: suggestion, discussion, refinement with humans
        \item Review each step as it happens
        \item E.g. copilot chat
    \end{itemize}

    \textbf{Delegation}
    \begin{itemize}
        \item Engineer: state desired outcome
        \item AI: Plan $\rightarrow$ execute (update filse, test, fix bugs, iterate)
        $\rightarrow$ present for review
        \item Review final output, not steps
        \item E.g. claude code, cursor, ...
    \end{itemize}

    \textbf{Orchestration}
    \begin{itemize}
        \item Several (specialized) agents, several roles
        \item AI: task decomposition, planning and execution
        \item Human or AI: Coordination
        \item E.g. CrewAI, LangGraph, ...
    \end{itemize}

    \textbf{Boosting productivity}
    \begin{itemize}
        \item Code efficiently thorugh autocomplete and boilerplate code generation, 
        allows for devs to try various approaches quickly
        \item Tackle new problems creatively by generating and explaining code, configs and docs
        \item Code translation (e.g. legacy code to python)
        \item Writing maintainable code (detect code vulnerabilities and fix, improve code quality, enforce good practices)
        \item Better testing by generating tests and improving test coverage
        \item Agentic coding to automatically fix bugs (95+\% of GitHub issues resolved automatically)
    \end{itemize}

    \textbf{Potential risks}

    \begin{enumerate}
        \item Garbage In Garbage Out (output is only as good as input)
        \begin{itemize}
            \item Regurgitation of bad code
            \item More applicable to more niche use-cases
            \item Correctness is not guaranteed $\Rightarrow$ reduced trust (e.g. policy for db tuning needs validation)
        \end{itemize}
        \item Traceability of the source
        \begin{itemize}
            \item Not all code can be used as is (restrictions: license terms, correctness, etc.)
            \item Recourse: some tools claim to flag training data (e.g. Amazon CodeWhisperer)
        \end{itemize}
        \item Data safety
        \begin{itemize}
            \item Confidential data may be leaked (e.g. Samsung bans ChatGPT)
        \end{itemize}
        \item Susceptible to vulnerabilities
        \begin{itemize}
            \item Bad actors can plant malicious code in public domain that is used for training
        \end{itemize}
        \item Increasing technical debt
        \begin{itemize}
            \item Maintaining others' code is hard, maintaining automatically generated code is harder
        \end{itemize}
    \end{enumerate}

    \textbf{Vibe coding and technical debt}

    \uline{Benefits:}

    \begin{itemize}
        \item Compressed distance between:
        \begin{itemize}
            \item idea $\rightarrow$ prototype
            \item requirement $\rightarrow$ interface
            \item prompt $\rightarrow$ implementation
            \item feedback $\rightarrow$ revised feature
        \end{itemize}
        \item Resulting in rapid prototyping and MVP development
        \item Lower barriers for non-specialists and domain experts
        \item Faster exploration of alternative designs
    \end{itemize}

    \uline{Problems:}

    \begin{itemize}
        \item Inconsistent architecture
        \item Security / performance / maintainability $\downarrow$
        \item Increased review / verification / rework burden
    \end{itemize}

    Role of SWE if AI can do everyt:
    \begin{itemize}
        \item System engineer
        \begin{itemize}
            \item Understand user needs, decipher requirements
            \item Identify architecture decomposition and design
            \item Evaluate trade-offs, plan long-term evolution
        \end{itemize}
        \item Orchestrator
        \begin{itemize}
            \item Decompose problems, identify and provide contexts
            \item Guide AI tools and agents, coordinate coding-testing-review-documentation
        \end{itemize}
        \item Quality guardian
        \begin{itemize}
            \item Review and validate code correctness and security
            \item Manage technical debt and maintainability
            \item Monitor system performance
        \end{itemize}
        \item Skills that become more valuable: problem framing, system thinking, communication,

    \end{itemize}


    \uline{\textbf{Lecture 3 - Requirements}}

    Requirements are a set of capabilities and constraints

    [IEEE]
    \begin{enumerate}
        \item A condition or capability needed by a user to solve a problem or achieve an objective
        \item A condition or capability that must be met or possessed by a system or system component
        to satisfy a contract, standard, specification, or other formally imposed documents
        \item A documented representation fo a condition or capability as in (1) or (2)
    \end{enumerate}

    \textbf{Example of capability:} system should allow a requester to specify pick-up and drop-off locations.

    \textbf{Example of constraint:} pick-up and drop-off locations should be from a list of on-campus
    buildings / landmarks.

    \textbf{Requirements Development Phases}
    \begin{Figure}
        \centering
        \includegraphics[width=\linewidth]{requirements_development_phases.png}
    \end{Figure}
    \textbf{Elicitation}: Discover requirements (through interview, workshop, document analysis, prototyping)

    \textbf{Analysis}: Analyze, decompose, derive, understand, negotiate requirements, identify gaps

    \textbf{Specification}: Written and illustrated requirements for comprehension, review, and use

    \textbf{Validation}: Confirm correct set of requirements that will enable developers to build a solution

    \textbf{Sources of software requirements}
    \begin{itemize}
        \item Documents
        \begin{itemize}
            \item Descriptions of current or competing products
            \item Standards or regulations
            \item help desk problem reports
        \end{itemize}
        \item Interviews
        \begin{itemize}
            \item Focus groups
            \item Use case workshops
            \item Product champions as key customer rep
        \end{itemize}
        \item Questionnaires and marketing surveys
        \begin{itemize}
            \item Run a pilot first - difficult to ask clear questions
        \end{itemize}
        \item Event-response tables
        \begin{itemize}
            \item Identify external stimuli and describe system responses
        \end{itemize}
        \item Observation - watch users do their jobs
        \begin{itemize}
            \item Create workflow diagrams
            \item "Day in the life" studies
            \item Look at what information the user has when performing a particular task
        \end{itemize}
        \item Prototyping
        \begin{itemize}
            \item Useful for requirements, design, and implementation
        \end{itemize}
    \end{itemize}
    \textbf{FR vs NFR}

    \textbf{FR} describes something the system should do
    \begin{itemize}
        \item specify a behavior or function
        \item e.g. the passenger shall be able to print boarding passes for all flight segments for which she has checked in
    \end{itemize}
    \textbf{NFR} describes something not directly related to the functionality of the system
    \begin{itemize}
        \item May describe how well the system works
        \item Can include other attributes too (e.g. reliability, efficiency, usability)
        \item E.g. mean time between failure $\geq$ 9000 hrs
    \end{itemize}

    \textbf{External vs Internal Quality Attributes}

    \textbf{External:}

    \begin{itemize}
        \item Observed when the software is executing
        \item Impacts user's experienece of using the software
        \item Develops the user's perception of software quality
        \item E.g. Availability, performance, robustness, safety, security, reliability, deployability, compatibility, installability, usability, interoperability, integrity
    \end{itemize}

    \textbf{Internal:}

    \begin{itemize}
        \item Not directly observed when software is executing
        \item Perceived by the developers and maintainers
        \item Encompasses aspects of design that may impact external attributes
        \item E.g. Scalability, efficiency, modifiability, verifiability, portability, maintainability, reusability, testability
    \end{itemize}

    \uline{\textbf{Lecture 4 - Software Architecture}}

    The \textbf{Software Architecture} of a program or computing system is the \textbf{structure or structures of the system},
    which comprise software components, the \textbf{externally visible properties} of those components, and the relationships between them.

    (Externally visible properties = FR + NFR)
    
    In short, the overall organization of the system and a high level design

    \textbf{Building blocks of architecture}
    \begin{itemize}
        \item Component: Element that models an application-specific function, responsibility, requirement, task, or process
        \item Connector: Element that models interactions among components for the purpose of transfer of control and/or data
        \item Configuration: Topology or structure
    \end{itemize}

    \textbf{Message and Event}
    \begin{itemize}
        \item Components communiacte via event notifications / message passing / RPC and other protocols
        \item Async and sync
        \item Point-to-point
    \end{itemize}
    A message is some data sent to a specific address.
    \begin{itemize}
        \item In message-driven systems, each component has a unique address to which other components can send messages
        \item Each component, or recipient, awaits messages and reacts to them
    \end{itemize}
    An event is some data emitted from a component for anyone listening to consume
    \begin{itemize}
        \item An event is emitted by a component when its state changes - record of something that has happened 
        \item They are immutable (cannot be changed or deleted)
        \item They are ordered in the sequence of their creation
        \item Sent to the publishing infrastructure, where consumers may later retrieve it
    \end{itemize}

    \textbf{Principle of Modularity}

    \begin{itemize}
        \item Modularization $\rightarrow$ shorter development time, better flexibility, better comprehensibility
        \item Decomposing a big chunk into smaller chunks with well-defined interfaces (APIs)
        \item A way to manage the complexity of a problem by breaking them down to smaller manageable modules 
    \end{itemize}

    \textbf{Types of cohesion}

    \begin{itemize}
        \item Functional
        \begin{itemize}
            \item Facilities perform only one computation with no side effects
        \end{itemize}
        \item Layer
        \begin{itemize}
            \item Related services are kept together, everything else is kept out, and there is a strict hierarchy in which higher-level services can access only lower-level services. Accessing a service may result in side effects
        \end{itemize}
        \item Communicational
        \begin{itemize}
            \item Facilities for operating ont he same data are kept together
        \end{itemize}
        \item Sequential
        \begin{itemize}
            \item A set of procedures, which work in sequence to perform some computation, are kept together. Output is from one input to the next.
        \end{itemize}
        \item Procedural
        \begin{itemize}
            \item A set of procedures, which are called one after another, are kept together
        \end{itemize}
        \item Temporal
        \begin{itemize}
            \item Procedures used in the same general phase of execution, such as initialization or termination, are kept together
        \end{itemize}
        \item Utility
        \begin{itemize}
            \item Related utilities are kept together, when there is no way to group them using a stronger form of cohesion
        \end{itemize}
    \end{itemize}

    \textbf{Types of coupling}
    \begin{itemize}
        \item Content
        \begin{itemize}
            \item Component modifies internal data of another
        \end{itemize}
        \item Common / global
        \begin{itemize}
            \item Using global variables; all modules using the global variable are coupled
        \end{itemize}
        \item Control
        \begin{itemize}
            \item One module (A) directly controls the other (B) by passing information on what B should do; e.g., using a ``flag'' that defines what B should do
        \end{itemize}
        \item Data
        \begin{itemize}
            \item One module sharing data with another module; e.g., via passing parameters
        \end{itemize}
        \item External
        \begin{itemize}
            \item A dependency exists to elements outside the scope of the system, such as the operating system, shared libraries or the hardware.
        \end{itemize}
        \item Temporal
        \begin{itemize}
            \item Two actions are bundled together just because they happen to occur at the same time
        \end{itemize}
        \item Inclusion / import
        \begin{itemize}
            \item Including a tile or importing a package
        \end{itemize}
    \end{itemize}

    \textbf{Architecture styles}
    \begin{enumerate}
        \item How is the code divided?
        \begin{itemize}
            \item Technical concern (horizontal)
            \begin{itemize}
                \item Focuses on the \textbf{separation of concerns}
                \item May align with the team's expertise (e.g. frontend, backend, db administrator)
            \end{itemize}
            \item Domain (business concern) (vertical)
            \begin{itemize}
                \item Aligned with the domain (business concern)
                \item Logical components map to the problem (e.g. customer, payment, shipping)
            \end{itemize}
        \end{itemize}
        \item How is the system deployed?
        \begin{itemize}
            \item Single deployed unit (monolithic)
            \begin{itemize}
                \item Deploy all logical components that make up the application as one unit
                \item Application runs as one process
            \end{itemize}
            \item Distributed units
            \begin{itemize}
                \item Application consists of independent logical components
                \item Logical components run as individual processes
                \item Communicate over the network
            \end{itemize}
        \end{itemize}
    \end{enumerate}

            \item Improved fault isolation

        \end{itemize}
    \end{itemize}
